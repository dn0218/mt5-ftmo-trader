# ============================================================
# mt5-ftmo-trader 配置模板

# ── MT5 连接（必填三件套）────────────────────────────
MT5_LOGIN=你的MT5登录号
MT5_PASSWORD=你的MT5密码
MT5_SERVER=你的经纪商服务器名称（如 ICMarkets-Demo）

# ── LLM 提供商选择：openai 或 ollama ──────────────────

LLM_ENABLED=true
LLM_TIMEOUT=30

# ── 方案一：OpenAI 兼容 API（接自有模型）───────────────
# ⚠️ OPENAI_BASE_URL 依赖已应用的补丁（services/decision/ollama_decider.py）
OPENAI_API_KEY=你的API_Key
# 填你的服务根地址，通常以 /v1 结尾，末尾不要带 /chat/completions
OPENAI_BASE_URL=https://your-api-endpoint.example.com/v1
OPENAI_MODEL=你的模型名称（如 qwen2.5-72b-instruct）

OLLAMA_TIMEOUT=60
